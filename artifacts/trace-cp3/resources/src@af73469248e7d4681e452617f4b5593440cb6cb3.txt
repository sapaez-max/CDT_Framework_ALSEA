import { expect, test, type Page, type TestInfo } from '@fixtures/base.fixture';
import { LoginPage } from '@pages/auth/LoginPage';
import { MenuAdministrationPage } from '@pages/menu/MenuAdministrationPage';
import { findEditedTemplate } from '@utils/template-editor';
import type { FilterLoadCase } from './casos.data';
import { goToLanding } from './ordenamiento-flow';

export async function ejecutarCargaFiltros(
  page: Page,
  caseData: FilterLoadCase,
  testInfo: TestInfo,
): Promise<void> {
  await goToLanding(page);
  await new LoginPage(page).expectAuthenticated();

  const editedTemplatePath = await test.step(
    `Localizar plantilla editada generada por ${caseData.sourceCaseId}`,
    async () => findEditedTemplate(caseData.sourceCaseId),
  );

  testInfo.annotations.push(
    { type: 'Plantilla editada origen', description: editedTemplatePath },
    { type: 'Caso origen de edicion', description: caseData.sourceCaseId },
  );

  const menuPage = new MenuAdministrationPage(page);
  await menuPage.openFilterLoad();
  await menuPage.loadFilters(caseData, editedTemplatePath);

  await test.step('Adjuntar referencia de la plantilla usada para carga de filtros', async () => {
    await expect(editedTemplatePath, 'La plantilla cargada debe ser un archivo Excel').toMatch(/\.xlsx?$/i);
    await testInfo.attach(`plantilla-carga-filtros-${caseData.id}`, {
      path: editedTemplatePath,
      contentType: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
    });
  });
}
